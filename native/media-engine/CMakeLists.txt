cmake_minimum_required(VERSION 3.24)
project(darling_media VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Darling media engine currently requires Windows")
endif()
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 60 INACTIVITY_TIMEOUT 20)
FetchContent_MakeAvailable(json)
add_executable(darling-media src/main.cpp src/devices.cpp src/capture.cpp src/playback.cpp)
target_compile_features(darling-media PRIVATE cxx_std_20)
target_compile_definitions(darling-media PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(darling-media PRIVATE nlohmann_json::nlohmann_json ole32 propsys uuid avrt)
if(MSVC)
  target_compile_options(darling-media PRIVATE /W4 /EHsc /utf-8)
endif()
install(TARGETS darling-media RUNTIME DESTINATION .)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json-LICENSE.MIT)
enable_testing()
add_executable(speech-wav-test tests/speech_wav_test.cpp)
target_compile_features(speech-wav-test PRIVATE cxx_std_20)
add_test(NAME speech-wav COMMAND speech-wav-test)
option(DARLING_AEC3_EXPERIMENT "Build isolated AEC3 evaluation (not enabled in the client)" OFF)
if(DARLING_AEC3_EXPERIMENT)
  include(cmake/aec3.cmake)
endif()
add_executable(level-meter-test tests/level_meter_test.cpp)
target_compile_features(level-meter-test PRIVATE cxx_std_20)
target_include_directories(level-meter-test PRIVATE src)
add_test(NAME level-meter COMMAND level-meter-test)
add_executable(audio-pipeline-test tests/audio_pipeline_test.cpp)
target_compile_features(audio-pipeline-test PRIVATE cxx_std_20)
target_include_directories(audio-pipeline-test PRIVATE src)
add_test(NAME audio-pipeline COMMAND audio-pipeline-test)
